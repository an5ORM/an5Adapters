cmake_minimum_required(VERSION 3.16)
project(an5_vector LANGUAGES C)
find_path(SQLITE_INCLUDE_DIR sqlite3ext.h REQUIRED)
add_library(an5_vector MODULE an5_vector.c)
target_include_directories(an5_vector PRIVATE "${SQLITE_INCLUDE_DIR}")
set_target_properties(an5_vector PROPERTIES PREFIX "" C_STANDARD 99)
if(MSVC)
  target_compile_options(an5_vector PRIVATE /O2 /W4)
else()
  target_compile_options(an5_vector PRIVATE -O3 -Wall -Wextra -Werror)
  target_link_libraries(an5_vector PRIVATE m)
endif()
